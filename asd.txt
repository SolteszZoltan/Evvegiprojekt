<script>
function showSection(id) {
    document.querySelectorAll('.section').forEach(sec => sec.classList.remove('active'));
    document.getElementById(id).classList.add('active');
}

async function loadRaktar() {
    try {
        const res = await fetch('http://localhost:3000/api/raktar');
        const data = await res.json();

        const tbody = document.querySelector('#raktarTable tbody');
        tbody.innerHTML = '';

        data.forEach(item => {
            tbody.innerHTML += `<tr><td>${item.nev}</td><td>${item.mennyiseg}</td></tr>`;
        });
    } catch (err) {
        console.error(err);
        alert('Nem sikerült betölteni az adatokat');
    }
}

loadRaktar();
</script>
